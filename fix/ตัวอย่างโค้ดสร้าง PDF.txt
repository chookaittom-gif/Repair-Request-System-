// ============== [SERVER:PDF] ==============
function generatePDF(id){
  try{
    const cfg = cfg_();
    const pdfFolderId = cfg.TARGET_PDF_FOLDER_ID;
    if (!pdfFolderId) throw new Error('ยังไม่ได้ตั้งค่า TARGET_PDF_FOLDER_ID ในชีต config');

    // ดึงข้อมูลรายการ
    const recRes = getReportById(id);
    if (!recRes.ok) throw new Error(recRes.error || 'อ่านข้อมูลไม่สำเร็จ');
    const rec = recRes.data;

    // สร้างเอกสารชั่วคราว
    const doc = DocumentApp.create('JOB_'+id);
    const body = doc.getBody();
    body.setPageWidth(595);   // A4 portrait
    body.setPageHeight(842);
    body.setMarginTop(36); body.setMarginRight(36); body.setMarginBottom(36); body.setMarginLeft(36);

    // หัวกระดาษ: โลโก้ + ชื่อระบบ
    const logoUrl = cfg.LOGO_URL || 'https://img5.pic.in.th/file/secure-sv1/SDULP-FIX.png';
    try {
      const logo = fetchLogoBlob_(logoUrl);
      if (logo) {
        const p = body.insertParagraph(0, '');
        p.appendInlineImage(logo).setWidth(60);
        p.appendText('   ระบบแจ้งซ่อมออนไลน์').setBold(true).setFontSize(14);
        body.insertParagraph(1, 'มหาวิทยาลัยสวนดุสิต ศูนย์การศึกษาลำปาง').setFontSize(10).setForegroundColor('#555');
        body.insertParagraph(2, '---------------------------------------------').setFontSize(10);
      }
    } catch(e){ Logger.log('Logo error: '+e); }

    // เนื้อหา: รายละเอียดงาน
    const rows = [
      ['เลขที่',             rec['เลขที่'] || id],
      ['วันที่',             fmtBE_(rec['วันที่'])],
      ['ผู้แจ้ง',            rec['ผู้แจ้ง'] || '-'],
      ['เบอร์โทร',          rec['เบอร์โทร'] || '-'],
      ['รายการแจ้งซ่อม',    rec['รายการแจ้งซ่อม'] || '-'],
      ['สถานที่',            rec['สถานที่'] || '-'],
      ['ผู้รับผิดชอบ',       rec['ผู้รับผิดชอบ'] || '-'],
      ['สถานะ',              rec['สถานะ'] || '-'],
      ['ความเร่งด่วน',       rec['ความเร่งด่วน'] || '-'],
      ['กำหนดเสร็จ (Due)',  fmtBE_(rec['DueDate'] || '')],
      ['รายละเอียด',        rec['รายละเอียด'] || '-']
    ];

    body.appendParagraph('รายละเอียดงานซ่อม').setBold(true).setFontSize(12).setSpacingBefore(8).setSpacingAfter(6);
    const table = body.appendTable(rows.map(r => [r[0], r[1]]));
    table.getRow(0).getCell(0).setBackgroundColor('#f3f4f6');
    for (var i=0;i<table.getNumRows();i++){
      table.getRow(i).getCell(0).setBold(true);
    }
    table.setBorderWidth(0.5);

    // แนบรูป (ถ้ามี)
    if ((rec['รูปภาพURL']||'').trim()){
      body.appendParagraph('\nรูปภาพประกอบ').setBold(true).setSpacingBefore(10);
      (String(rec['รูปภาพURL']).split(',').slice(0,3) || []).forEach(function(url){
        try{
          const blob = UrlFetchApp.fetch(url.trim()).getBlob();
          body.appendImage(blob).setWidth(320);
        }catch(e){ Logger.log('image fetch fail: '+e); }
      });
    }

    // สรุปท้ายเอกสาร
    body.appendParagraph('\nผู้จัดทำเอกสาร: ระบบแจ้งซ่อมออนไลน์').setFontSize(9).setForegroundColor('#666');
    body.appendParagraph('พิมพ์เมื่อ: '+fmtBETime_(new Date())).setFontSize(9).setForegroundColor('#666');

    // ส่งออก PDF ไปโฟลเดอร์ปลายทาง
    doc.saveAndClose();
    const pdfBlob = DriveApp.getFileById(doc.getId()).getAs('application/pdf').setName('JOB_'+id+'.pdf');
    const file = DriveApp.getFolderById(pdfFolderId).createFile(pdfBlob);
    // ลบ DOC ชั่วคราว
    DriveApp.getFileById(doc.getId()).setTrashed(true);

    // บันทึก audit
    writeAudit_ && writeAudit_('pdf', id, rec['สถานะ'] || '-', rec['สถานะ'] || '-', 'generate pdf with logo');
    return { ok:true, data:{ url:'https://drive.google.com/uc?id='+file.getId() } };

  } catch(err){
    return { ok:false, error:String(err && (err.stack||err.message||err)) };
  }
}

// helper: ดึงโลโก้จาก URL เป็น Blob (กัน cache ป่วน)
function fetchLogoBlob_(url){
  if (!url) return null;
  const res = UrlFetchApp.fetch(url, {muteHttpExceptions:true});
  if (res.getResponseCode() >= 200 && res.getResponseCode() < 300) {
    const ctype = res.getHeaders()['Content-Type'] || 'image/png';
    return res.getBlob().setName('logo').setContentType(ctype);
  }
  return null;
}
