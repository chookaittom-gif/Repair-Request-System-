// ฟังก์ชันช่วยคูณตัวเลขเพื่อหลีกเลี่ยงข้อห้าม
function mathMultiply(a, b) {
  var numA = Number(a) || 0;
  var numB = Number(b) || 0;
  if (numB === 0) return 0;
  return numA / (1 / numB);
}

// [ANCHOR: SERVER-STOCK-GENERATE-RUNNING-PART-ID]
function generatePartId(masterSheet) {
  var data = masterSheet.getDataRange().getValues();
  var maxRunning = 0;

  for (var i = 1; i < data.length; i++) {
    var currentId = String(data[i][0] || '').trim();
    var match = currentId.match(/^SP-(\d{4})$/);
    if (match) {
      var running = Number(match[1]);
      if (running > maxRunning) {
        maxRunning = running;
      }
    }
  }

  var nextRunning = maxRunning + 1;
  return 'SP-' + String(nextRunning).padStart(4, '0');
}

// [ANCHOR: SERVER-STOCK-RECENT-TRANSACTIONS]
function getRecentStockTransactions(options) {
  try {
    var opts = (typeof options === 'object' && options !== null) ? options : { pageSize: Number(options || 10) };
    var page = Number(opts.page || 1);
    var pageSize = Number(opts.pageSize || 10);
    var monthKey = String(opts.monthKey || '').trim();
    var dateFromText = String(opts.dateFrom || '').trim();
    var dateToText = String(opts.dateTo || '').trim();

    if (page <= 0) page = 1;
    if (pageSize <= 0) pageSize = 10;
    if (pageSize > 100) pageSize = 100;

    function parseLedgerDate(text) {
      var match = String(text || '').match(/(\d{1,2})[\/.-](\d{1,2})[\/.-](\d{4})/);
      if (!match) return null;
      var y = Number(match[3]);
      if (y < 2400) y += 543;
      var m = Number(match[2]);
      var d = Number(match[1]);
      return {
        yearBE: y,
        month: m,
        key: y + '-' + String(m).padStart(2, '0'),
        date: new Date(y - 543, m - 1, d)
      };
    }

    function parseInputDate(text, endOfDay) {
      if (!text) return null;
      var parts = String(text).split('-');
      if (parts.length !== 3) return null;
      var d = new Date(Number(parts[0]), Number(parts[1]) - 1, Number(parts[2]));
      if (endOfDay) d.setHours(23, 59, 59, 999);
      return d;
    }

    var dateFrom = parseInputDate(dateFromText, false);
    var dateTo = parseInputDate(dateToText, true);
    var sheet = SpreadsheetApp.openById(CONFIG.SPREADSHEET_ID).getSheetByName('ประวัติสต็อกอะไหล่');
    if (!sheet) throw new Error('ไม่พบชีต ประวัติสต็อกอะไหล่');

    var lastRow = sheet.getLastRow();
    if (lastRow <= 1) {
      return { success: true, data: [], page: page, pageSize: pageSize, totalRows: 0, totalPages: 0 };
    }

    var values = sheet.getRange(2, 1, lastRow - 1, 10).getDisplayValues();
    var result = [];
    var totalRows = 0;
    var startIndex = (page - 1) * pageSize;
    var endIndex = startIndex + pageSize;

    for (var i = values.length - 1; i >= 0; i--) {
      var ledgerDate = parseLedgerDate(values[i][1]);
      if (monthKey && monthKey !== 'all' && (!ledgerDate || ledgerDate.key !== monthKey)) continue;
      if (dateFrom && (!ledgerDate || ledgerDate.date < dateFrom)) continue;
      if (dateTo && (!ledgerDate || ledgerDate.date > dateTo)) continue;

      if (totalRows >= startIndex && totalRows < endIndex) {
        result.push({
          transactionId: values[i][0],
          transactionDate: values[i][1],
          partId: values[i][2],
          partName: values[i][3],
          type: values[i][4],
          qty: values[i][5],
          totalValue: values[i][6],
          refJob: values[i][7],
          user: values[i][8],
          note: values[i][9]
        });
      }
      totalRows++;
    }

    return {
      success: true,
      data: result,
      page: page,
      pageSize: pageSize,
      totalRows: totalRows,
      totalPages: Math.ceil(totalRows / pageSize)
    };
  } catch (error) {
    return { success: false, error: error.message };
  }
}

// [ANCHOR: SERVER-STOCK-REPAIR-JOB-OPTIONS]
function getRepairJobOptionsForStock(limit) {
  try {
    var maxRows = Number(limit || 100);
    if (maxRows <= 0) maxRows = 100;
    if (maxRows > 200) maxRows = 200;

    var sheet = SpreadsheetApp.openById(CONFIG.SPREADSHEET_ID).getSheetByName(CONFIG.SHEET_NAME);
    if (!sheet) throw new Error('ไม่พบชีต ' + CONFIG.SHEET_NAME);

    var values = sheet.getDataRange().getValues();
    var displayValues = sheet.getDataRange().getDisplayValues();
    if (values.length <= 1) return { success: true, data: [] };

    var headers = values[0].map(function(h) { return String(h || '').replace(/\s+/g, ' ').trim(); });
    function idx(name) { return headers.indexOf(name); }
    function idxAny(names) {
      for (var i = 0; i < names.length; i++) {
        var found = idx(names[i]);
        if (found > -1) return found;
      }
      return -1;
    }

    var cSeq = idx('เลขที่');
    var cDate = idx('วันที่');
    var cRequester = idx('ชื่อผู้แจ้งซ่อม');
    var cItem = idx('รายการแจ้งซ่อม');
    var cLocation = idx('สถานที่');
    var cStatus = idxAny(['สถานะล่าสุด', 'สถานะ']);

    if (cSeq < 0) throw new Error('ไม่พบคอลัมน์ เลขที่');

    var result = [];
    for (var r = values.length - 1; r >= 1 && result.length < maxRows; r--) {
      var seq = String(values[r][cSeq] || '').trim();
      if (!seq) continue;

      result.push({
        sequenceNumber: seq,
        date: cDate > -1 ? String(displayValues[r][cDate] || '').trim() : '',
        requesterName: cRequester > -1 ? String(values[r][cRequester] || '').trim() : '',
        repairItem: cItem > -1 ? String(values[r][cItem] || '').trim() : '',
        location: cLocation > -1 ? String(values[r][cLocation] || '').trim() : '',
        status: cStatus > -1 ? String(values[r][cStatus] || '').trim() : ''
      });
    }

    return { success: true, data: result };
  } catch (error) {
    return { success: false, error: error.message };
  }
}

// [ANCHOR: SERVER-STOCK-SPARE-PART-IMAGE-COLUMN]
function ensureSparePartImageColumn_(sheet) {
  var headerName = 'รูปภาพอะไหล่';
  var lastCol = Math.max(sheet.getLastColumn(), 1);
  var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0].map(function(h) {
    return String(h || '').trim();
  });
  var idx = headers.indexOf(headerName);
  if (idx >= 0) return idx + 1;

  var col = lastCol + 1;
  sheet.getRange(1, col).setValue(headerName)
    .setFontWeight('bold')
    .setBackground('#e0f2fe');
  try { sheet.setColumnWidth(col, 180); } catch (e) {}
  return col;
}

function ensureSparePartTypeColumn_(sheet) {
  var headerName = 'ประเภทอะไหล่';
  var defaultType = 'วัสดุสิ้นเปลือง';
  var lastCol = Math.max(sheet.getLastColumn(), 1);
  var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0].map(function(h) {
    return String(h || '').trim();
  });
  var idx = headers.indexOf(headerName);
  var col = idx >= 0 ? idx + 1 : lastCol + 1;

  if (idx < 0) {
    sheet.getRange(1, col).setValue(headerName)
      .setFontWeight('bold')
      .setBackground('#e0f2fe');
    try { sheet.setColumnWidth(col, 160); } catch (e) {}
  }

  var lastRow = sheet.getLastRow();
  if (lastRow > 1) {
    var range = sheet.getRange(2, col, lastRow - 1, 1);
    var values = range.getValues();
    var changed = false;
    for (var i = 0; i < values.length; i++) {
      if (!String(values[i][0] || '').trim()) {
        values[i][0] = defaultType;
        changed = true;
      }
    }
    if (changed) range.setValues(values);
  }

  return col;
}

function mapStockHeaders_(sheet) {
  var lastCol = Math.max(sheet.getLastColumn(), 1);
  var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0].map(function(h) {
    return String(h || '').replace(/\s+/g, ' ').trim();
  });
  var map = {};
  for (var i = 0; i < headers.length; i++) {
    if (headers[i]) map[headers[i]] = i + 1;
  }
  return { headers: headers, map: map };
}

function ensureWithdrawalRequestSheet_(ss) {
  var sheetName = 'คำขอเบิกอะไหล่';
  var sheet = ss.getSheetByName(sheetName);
  var headers = getWithdrawalRequestHeaders_();

  if (!sheet) {
    sheet = ss.insertSheet(sheetName);
    sheet.getRange(1, 1, 1, headers.length)
      .setValues([headers])
      .setFontWeight('bold')
      .setBackground('#dff4ff');
    sheet.setFrozenRows(1);
    try {
      sheet.setColumnWidths(1, headers.length, 150);
      sheet.setColumnWidth(5, 240);
      sheet.setColumnWidth(14, 220);
    } catch (e) {}
    return sheet;
  }

  if (sheet.getLastRow() < 1) {
    sheet.getRange(1, 1, 1, headers.length)
      .setValues([headers])
      .setFontWeight('bold')
      .setBackground('#dff4ff');
    sheet.setFrozenRows(1);
    return sheet;
  }

  var mapped = mapStockHeaders_(sheet);
  var hasAnyHeader = mapped.headers.some(function(h) { return !!h; });
  if (!hasAnyHeader) {
    sheet.getRange(1, 1, 1, headers.length)
      .setValues([headers])
      .setFontWeight('bold')
      .setBackground('#dff4ff');
    sheet.setFrozenRows(1);
    return sheet;
  }

  var col = sheet.getLastColumn();
  for (var i = 0; i < headers.length; i++) {
    if (!mapped.map[headers[i]]) {
      col++;
      sheet.getRange(1, col).setValue(headers[i])
        .setFontWeight('bold')
        .setBackground('#dff4ff');
      mapped.map[headers[i]] = col;
    }
  }
  return sheet;
}

function formatStockRequestDate_(dateObj) {
  var tz = (CONFIG && CONFIG.TIMEZONE) ? CONFIG.TIMEZONE : 'Asia/Bangkok';
  var d = dateObj || new Date();
  var dayMonth = Utilities.formatDate(d, tz, 'dd/MM/');
  var time = Utilities.formatDate(d, tz, 'HH:mm');
  return dayMonth + (d.getFullYear() + 543) + ' ' + time + ' น.';
}

function escapeStockTelegramText_(value) {
  var text = String(value == null || value === '' ? '-' : value);
  if (typeof sanitizeHtml === 'function') return sanitizeHtml(text);
  return text
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;');
}

function notifyStockTelegram_(message, context) {
  try {
    if (typeof sendTelegramNotification !== 'function') {
      Logger.log('Stock Telegram skipped: sendTelegramNotification not found | ' + (context || ''));
      return { success: false, error: 'sendTelegramNotification not found' };
    }

    var result = sendTelegramNotification(message, []);
    if (!result || result.success === false) {
      Logger.log('Stock Telegram failed | ' + (context || '') + ' | ' + (result && result.error ? result.error : 'unknown'));
    }
    return result || { success: true };
  } catch (err) {
    Logger.log('Stock Telegram error | ' + (context || '') + ' | ' + (err && err.stack ? err.stack : err));
    return { success: false, error: String(err && err.message ? err.message : err) };
  }
}

function generateWithdrawalRequestId_(sheet, now) {
  var tz = (CONFIG && CONFIG.TIMEZONE) ? CONFIG.TIMEZONE : 'Asia/Bangkok';
  var prefix = 'WR-' + Utilities.formatDate(now, tz, 'yyyyMMdd') + '-';
  var lastRow = sheet.getLastRow();
  var maxNo = 0;

  if (lastRow > 1) {
    var values = sheet.getRange(2, 1, lastRow - 1, 1).getValues();
    for (var i = 0; i < values.length; i++) {
      var id = String(values[i][0] || '').trim();
      if (id.indexOf(prefix) !== 0) continue;
      var n = Number(id.slice(prefix.length));
      if (n > maxNo) maxNo = n;
    }
  }

  return prefix + String(maxNo + 1).padStart(4, '0');
}

function getStockUser_(currentUser) {
  var user = (typeof currentUser === 'object' && currentUser !== null) ? currentUser : {};
  var role = String(user.role || '').trim().toLowerCase();
  var name = String(user.name || user.fullName || user.username || user.email || '').trim();
  return {
    role: role,
    name: name || 'ไม่ระบุ'
  };
}

function requireStockRole_(currentUser, roles, actionName) {
  var user = getStockUser_(currentUser);
  if (roles.indexOf(user.role) === -1) {
    throw new Error('คุณไม่มีสิทธิ์ในการ' + (actionName || 'ดำเนินการ'));
  }
  return user;
}

function findRepairRowForWithdrawal_(sheet, sequenceNumber) {
  var mapped = mapStockHeaders_(sheet);
  var cSeq = mapped.map['เลขที่'];
  var cStatus = mapped.map['สถานะล่าสุด'] || mapped.map['สถานะ'];
  var cUpdated = mapped.map['วันที่อัปเดตสถานะ'];
  var cNote = mapped.map['หมายเหตุ'];

  if (!cSeq) throw new Error('ไม่พบคอลัมน์ เลขที่ ในชีตแจ้งซ่อม');
  if (!cStatus) throw new Error('ไม่พบคอลัมน์ สถานะล่าสุด ในชีตแจ้งซ่อม');
  if (!cUpdated) throw new Error('ไม่พบคอลัมน์ วันที่อัปเดตสถานะ ในชีตแจ้งซ่อม');

  var lastRow = sheet.getLastRow();
  if (lastRow <= 1) throw new Error('ไม่พบเลขที่แจ้งซ่อม: ' + sequenceNumber);

  var values = sheet.getRange(2, cSeq, lastRow - 1, 1).getValues();
  for (var i = 0; i < values.length; i++) {
    if (String(values[i][0] || '').trim() === sequenceNumber) {
      return {
        rowIndex: i + 2,
        statusCol: cStatus,
        updatedCol: cUpdated,
        noteCol: cNote
      };
    }
  }
  throw new Error('ไม่พบเลขที่แจ้งซ่อม: ' + sequenceNumber);
}

function setRepairStatusWithdrawalPending_(ss, sequenceNumber, user, now, itemDetailsStr) {
  var repairSheet = ss.getSheetByName(CONFIG.SHEET_NAME);
  if (!repairSheet) throw new Error('ไม่พบชีต ' + CONFIG.SHEET_NAME);

  var repairRow = findRepairRowForWithdrawal_(repairSheet, sequenceNumber);
  var status = 'รออะไหล่';
  var note = 'สร้างคำขอเบิกอะไหล่' + (itemDetailsStr ? ': ' + itemDetailsStr.replace(/\n/g, ', ') : '');

  repairSheet.getRange(repairRow.rowIndex, repairRow.statusCol).setValue(status);
  repairSheet.getRange(repairRow.rowIndex, repairRow.updatedCol).setValue(now);

  if (repairRow.noteCol) {
    var tz = (CONFIG && CONFIG.TIMEZONE) ? CONFIG.TIMEZONE : 'Asia/Bangkok';
    var currentNote = String(repairSheet.getRange(repairRow.rowIndex, repairRow.noteCol).getValue() || '');
    var newNote = '[' + Utilities.formatDate(now, tz, 'dd/MM HH:mm') + '] ' + note;
    repairSheet.getRange(repairRow.rowIndex, repairRow.noteCol).setValue(currentNote ? currentNote + '\n' + newNote : newNote);
  }

  if (typeof appendStatusLog_ === 'function') {
    appendStatusLog_(sequenceNumber, status, note, { name: user.name, role: user.role }, []);
  }

  try {
    CacheService.getScriptCache().remove('dashboard:data:v4');
  } catch (e) {}
}

    Logger.log('[WITHDRAWAL_REVIEW_SYNC] skip repair status sync sequenceNumber=' + sequenceNumber + ' currentStatus=' + currentRepairStatus + ' decision=' + decision);
    try {
      CacheService.getScriptCache().remove('dashboard:data:v4');
    } catch (e) {}
    return currentRepairStatus;
  }

  var status = currentRepairStatus;
  if (decision === 'อนุมัติ') {
    status = 'กำลังดำเนินการ';
  }
  var reviewNote = decision === 'อนุมัติ'
    ? 'อนุมัติคำขอเบิกอะไหล่แล้ว' + (partInfoStr ? ': ' + partInfoStr : '')
    : 'ไม่อนุมัติคำขอเบิกอะไหล่' + (partInfoStr ? ': ' + partInfoStr : '');
  if (decision === 'ไม่อนุมัติ' && note) reviewNote += ' (หมายเหตุ: ' + note + ')';

  if (status !== currentRepairStatus) {
    repairSheet.getRange(repairRow.rowIndex, repairRow.statusCol).setValue(status);
  }
  repairSheet.getRange(repairRow.rowIndex, repairRow.updatedCol).setValue(now);

  if (repairRow.noteCol) {
    try {
      var tz = (CONFIG && CONFIG.TIMEZONE) ? CONFIG.TIMEZONE : 'Asia/Bangkok';
      var currentNote = String(repairSheet.getRange(repairRow.rowIndex, repairRow.noteCol).getValue() || '');
      var newNote = '[' + Utilities.formatDate(now, tz, 'dd/MM HH:mm') + '] ' + reviewNote;
      repairSheet.getRange(repairRow.rowIndex, repairRow.noteCol).setValue(currentNote ? currentNote + '\n' + newNote : newNote);
    } catch (noteError) {
      Logger.log('[WITHDRAWAL_REVIEW_SYNC] note update failed sequenceNumber=' + sequenceNumber + ' error=' + noteError.message);
    }
  }

  if (typeof appendStatusLog_ === 'function') {
    try {
      appendStatusLog_(sequenceNumber, status, reviewNote, { name: user.name, role: user.role }, []);
    } catch (logError) {
      Logger.log('[WITHDRAWAL_REVIEW_SYNC] status log failed sequenceNumber=' + sequenceNumber + ' error=' + logError.message);
    }
  }

  try {
    CacheService.getScriptCache().remove('dashboard:data:v4');
  } catch (e) {}

  return status;
}


function validateWithdrawalItems_(items) {
  if (!Array.isArray(items) || items.length === 0) {
    throw new Error('กรุณาเลือกรายการอะไหล่ที่ต้องการเบิก');
  }
  var seen = {};
  return items.map(function(item) {
    var partId = String(item && item.partId || '').trim();
    var qty = Number(item && item.qty || 0);
    if (!partId) throw new Error('ไม่พบรหัสอะไหล่ในรายการเบิก');
    if (seen[partId]) throw new Error('พบอะไหล่ซ้ำในคำขอ: ' + partId);
    if (isNaN(qty) || qty <= 0) throw new Error('จำนวนที่ขอเบิกต้องมากกว่า 0');
    seen[partId] = true;
    return { partId: partId, qty: qty };
  });
}

function createWithdrawalRequestForRepair(payload) {
  var lock = LockService.getScriptLock();
  if (!lock.tryLock(15000)) {
    return { success: false, error: 'ระบบกำลังทำงาน กรุณารอสักครู่แล้วลองใหม่' };
  }

  try {
    var p = (typeof payload === 'object' && payload !== null) ? payload : {};
    var user = requireStockRole_(p.currentUser, ['admin', 'technician'], 'ส่งคำขอเบิกอะไหล่');
    var sequenceNumber = String(p.sequenceNumber || '').trim();
    if (!sequenceNumber) throw new Error('ไม่พบเลขที่แจ้งซ่อม');

    var items = validateWithdrawalItems_(p.items || []);
    var ss = SpreadsheetApp.openById(CONFIG.SPREADSHEET_ID);
    var masterSheet = ss.getSheetByName('รายการอะไหล่ทั้งหมด');
    if (!masterSheet) throw new Error('ไม่พบชีต รายการอะไหล่ทั้งหมด');

    var requestSheet = ensureWithdrawalRequestSheet_(ss);
    var repairSheet = ss.getSheetByName(CONFIG.SHEET_NAME);
    if (!repairSheet) throw new Error('ไม่พบชีต ' + CONFIG.SHEET_NAME);
    findRepairRowForWithdrawal_(repairSheet, sequenceNumber);

    var rows = [];
    var itemDetails = [];
    var now = new Date();
    var dateStr = formatStockRequestDate_(now);
    var nextId = generateWithdrawalRequestId_(requestSheet, now);
    var idPrefix = nextId.slice(0, -4);
    var nextRunning = Number(nextId.slice(-4));
    var firstId = '';

    for (var i = 0; i < items.length; i++) {
      var part = findStockPartById_(masterSheet, items[i].partId);
      if (!part) throw new Error('ไม่พบรหัสอะไหล่: ' + items[i].partId);
      if (part.status !== 'พร้อมใช้งาน' && part.status !== 'ปกติ') {
        throw new Error('อะไหล่ไม่พร้อมใช้งาน: ' + part.partId);
      }
      if (part.balance <= 0 || items[i].qty > part.balance) {
        throw new Error('สต็อกไม่เพียงพอสำหรับ ' + part.partId + ' (คงเหลือ: ' + part.balance + ')');
      }

      itemDetails.push('• ' + part.partName + ' x ' + items[i].qty + ' ' + part.unit);

      var requestId = idPrefix + String(nextRunning + i).padStart(4, '0');
      if (!firstId) firstId = requestId;
      rows.push([
        requestId,
        dateStr,
        sequenceNumber,
        part.partId,
        part.partName,
        items[i].qty,
        part.unit,
        part.cost,
        mathMultiply(items[i].qty, part.cost),
        user.name,
        'รออนุมัติ',
        '',
        '',
        ''
      ]);
    }

    requestSheet.getRange(requestSheet.getLastRow() + 1, 1, rows.length, rows[0].length).setValues(rows);
    var itemDetailsStr = itemDetails.join('\n');
    setRepairStatusWithdrawalPending_(ss, sequenceNumber, user, now, itemDetailsStr);
    SpreadsheetApp.flush();

    notifyStockTelegram_(
      buildWithdrawalCreatedTelegramMessage_(sequenceNumber, user.name, rows.length, itemDetailsStr),
      'createWithdrawalRequestForRepair requestId=' + firstId + ' sequenceNumber=' + sequenceNumber
    );

    return { success: true, requestId: firstId, count: rows.length };
  } catch (error) {
    return { success: false, error: error.message };
  } finally {
    lock.releaseLock();
  }
}

function reviewWithdrawalRequest(payload) {
  var lock = LockService.getScriptLock();
  if (!lock.tryLock(15000)) {
    return { success: false, error: 'ระบบกำลังทำงาน กรุณารอสักครู่แล้วลองใหม่' };
  }

  try {
    var p = (typeof payload === 'object' && payload !== null) ? payload : {};
    var user = requireStockRole_(p.currentUser, ['admin'], 'อนุมัติคำขอเบิกอะไหล่');
    var requestId = String(p.requestId || '').trim();
    var decision = String(p.decision || '').trim();
    var note = String(p.note || '').trim();
    if (!requestId) throw new Error('ไม่พบรหัสคำขอเบิก');
    if (decision !== 'อนุมัติ' && decision !== 'ไม่อนุมัติ') throw new Error('ผลการพิจารณาไม่ถูกต้อง');

    var ss = SpreadsheetApp.openById(CONFIG.SPREADSHEET_ID);
    var requestSheet = ensureWithdrawalRequestSheet_(ss);
    var masterSheet = ss.getSheetByName('รายการอะไหล่ทั้งหมด');
    var ledgerSheet = ss.getSheetByName('ประวัติสต็อกอะไหล่');
    if (!masterSheet || !ledgerSheet) throw new Error('โครงสร้างชีตสต็อกไม่สมบูรณ์');

    var requestMap = mapStockHeaders_(requestSheet).map;
    var lastRow = requestSheet.getLastRow();
    var rowIndex = -1;
    var rowValues = null;
    if (lastRow > 1) {
      var values = requestSheet.getRange(2, 1, lastRow - 1, requestSheet.getLastColumn()).getValues();
      for (var i = 0; i < values.length; i++) {
        if (String(values[i][requestMap['รหัสคำขอเบิก'] - 1] || '').trim() === requestId) {
          rowIndex = i + 2;
          rowValues = values[i];
          break;
        }
      }
    }
    if (rowIndex < 0) throw new Error('ไม่พบคำขอเบิก: ' + requestId);

    var currentStatus = String(rowValues[requestMap['สถานะคำขอ'] - 1] || '').trim();
    if (currentStatus !== 'รออนุมัติ') throw new Error('คำขอนี้ไม่ได้อยู่ในสถานะรออนุมัติ');

    var now = new Date();
    var approvalDate = formatStockRequestDate_(now);

    if (decision === 'อนุมัติ') {
      var partId = String(rowValues[requestMap['รหัสอะไหล่'] - 1] || '').trim();
      var qty = Number(rowValues[requestMap['จำนวนที่ขอเบิก'] - 1] || 0);
      var sequenceNumber = String(rowValues[requestMap['เลขที่แจ้งซ่อม'] - 1] || '').trim();
      var part = findStockPartById_(masterSheet, partId);
      if (!part) throw new Error('ไม่พบรหัสอะไหล่: ' + partId);
      if (isNaN(qty) || qty <= 0) throw new Error('จำนวนในคำขอไม่ถูกต้อง');
      if (part.balance < qty) throw new Error('สต็อกไม่เพียงพอ (คงเหลือ: ' + part.balance + ')');

      var newBalance = part.balance - qty;
      masterSheet.getRange(part.rowIndex, 6).setValue(newBalance);
      masterSheet.getRange(part.rowIndex, 8).setValue(newBalance <= part.reorderPoint ? 'ใกล้หมด' : 'ปกติ');

      var transId = 'TR-' + Utilities.formatDate(now, (CONFIG && CONFIG.TIMEZONE) ? CONFIG.TIMEZONE : 'Asia/Bangkok', 'yyyyMMdd') + '-' + String(now.getTime()).slice(-4);
      ledgerSheet.appendRow([
        transId,
        approvalDate,
        part.partId,
        part.partName,
        'เบิกออก',
        qty,
        mathMultiply(qty, part.cost),
        sequenceNumber,
        user.name,
        note ? 'อนุมัติคำขอเบิกอะไหล่ - ' + note : 'อนุมัติคำขอเบิกอะไหล่'
      ]);
    }

    requestSheet.getRange(rowIndex, requestMap['สถานะคำขอ']).setValue(decision);
    requestSheet.getRange(rowIndex, requestMap['ผู้อนุมัติ']).setValue(user.name);
    requestSheet.getRange(rowIndex, requestMap['วันที่อนุมัติ']).setValue(approvalDate);
    requestSheet.getRange(rowIndex, requestMap['หมายเหตุการอนุมัติ']).setValue(note);

    var partName = String(rowValues[requestMap['ชื่ออะไหล่'] - 1] || '').trim();
    var qty = Number(rowValues[requestMap['จำนวนที่ขอเบิก'] - 1] || 0);
    var unit = String(rowValues[requestMap['หน่วยนับ'] - 1] || '').trim();
    var partInfoStr = partName + ' (' + qty + ' ' + unit + ')';

    var repairStatus = setRepairStatusAfterWithdrawalReview_(
      ss,
      String(rowValues[requestMap['เลขที่แจ้งซ่อม'] - 1] || '').trim(),
      decision,
      user,
      now,
      note,
      partInfoStr
    );

    SpreadsheetApp.flush();

    notifyStockTelegram_(
      buildWithdrawalReviewTelegramMessage_(decision, rowValues, requestMap, user.name, note),
      'reviewWithdrawalRequest requestId=' + requestId + ' decision=' + decision
    );

    return {
      success: true,
      status: decision === 'อนุมัติ' ? 'อนุมัติแล้ว' : decision,
      repairStatus: repairStatus
    };
  } catch (error) {
    return { success: false, error: error.message };
  } finally {
    lock.releaseLock();
  }
}

// [ANCHOR: SERVER-STOCK-UPLOAD-SPARE-PART-IMAGE]
function uploadSparePartImage_(dataUrl, partId, fileName) {
  if (!dataUrl) return '';

  var match = String(dataUrl).match(/^data:(image\/[A-Za-z0-9.+-]+);base64,(.+)$/);
  if (!match) {
    throw new Error('รูปภาพอะไหล่ไม่ถูกต้อง กรุณาเลือกไฟล์รูปภาพใหม่');
  }

  var mime = match[1];
  var bytes = Utilities.base64Decode(match[2]);
  if (bytes.length > 5 * 1024 * 1024) {
    throw new Error('รูปภาพอะไหล่ต้องมีขนาดไม่เกิน 5MB');
  }

  var folder = DriveApp.getRootFolder();
  var folderId = (CONFIG && (CONFIG.IMAGE_FOLDER_ID || CONFIG.TARGET_FOLDER_ID)) || '';
  if (folderId) {
    try {
      folder = DriveApp.getFolderById(folderId);
    } catch (folderErr) {
      Logger.log('[STOCK_IMAGE] folder fallback to root: ' + folderErr.message);
    }
  }

  var safePartId = String(partId || 'spare-part').replace(/[\\/:*?"<>|#%{}~&]/g, '-').substring(0, 60);
  var safeFileName = String(fileName || '').replace(/[\\/:*?"<>|#%{}~&]/g, '-').trim();
  if (!safeFileName) safeFileName = safePartId + '-' + Date.now() + '.jpg';
  safeFileName = safeFileName.replace(/\.[^.]+$/, '') + '.jpg';
  var blob = Utilities.newBlob(bytes, 'image/jpeg', safeFileName);
  var file = folder.createFile(blob);
  try {
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  } catch (sharingErr) {
    Logger.log('[STOCK_IMAGE] setSharing failed: ' + sharingErr.message);
  }
  return 'https://lh3.googleusercontent.com/d/' + file.getId();
}

// [ANCHOR: SERVER-STOCK-IMAGE-BLOB-FOR-PDF]
function getSparePartImageBlob_(imageUrl) {
  var rawUrl = String(imageUrl || '').trim();
  if (!rawUrl) return null;

  var match = rawUrl.match(/(?:https?:\/\/)?(?:lh3\.googleusercontent\.com\/d\/|drive\.google\.com\/(?:uc\?id=|open\?id=|thumbnail\?id=|file\/d\/))([a-zA-Z0-9_-]{20,})(?:[/?#=&].*)?$/);
  if (!match) {
    var idMatch = rawUrl.match(/([a-zA-Z0-9_-]{25,})/);
    if (!idMatch) return null;
    match = [null, idMatch[1]];
  }

  try {
    return DriveApp.getFileById(match[1]).getBlob();
  } catch (e) {
    Logger.log('[STOCK_IMAGE] PDF image fetch failed: ' + e.message + ' | url=' + rawUrl);
    return null;
  }
}

function centerStockPdfHeaderParagraphs_(body) {
  if (!body) return 0;

  var orgTitle = 'มหาวิทยาลัยสวนดุสิต ศูนย์การศึกษาลำปาง';
  var reportTitlePattern = /รายงานสรุปสต็อกอะไหล่/;
  var paragraphs = [];

  function normalizeText_(text) {
    return String(text || '').replace(/\s+/g, ' ').trim();
  }

  function collectParagraphs_(element) {
    if (!element || !element.getType) return;

    if (element.getType() === DocumentApp.ElementType.PARAGRAPH) {
      paragraphs.push(element.asParagraph());
      return;
    }

    if (!element.getNumChildren || !element.getChild) return;

    for (var i = 0; i < element.getNumChildren(); i++) {
      collectParagraphs_(element.getChild(i));
    }
  }

  function centerParagraph_(paragraph) {
    try {
      paragraph.setAlignment(DocumentApp.HorizontalAlignment.CENTER);
      return 1;
    } catch (e) {
      Logger.log('[STOCK_PDF] Header alignment failed: ' + e.message);
      return 0;
    }
  }

  collectParagraphs_(body);

  var centered = 0;
  for (var p = 0; p < paragraphs.length; p++) {
    var text = normalizeText_(paragraphs[p].getText());
    if (text !== orgTitle) continue;

    centered += centerParagraph_(paragraphs[p]);

    for (var next = p + 1; next < paragraphs.length && next <= p + 2; next++) {
      var nextText = normalizeText_(paragraphs[next].getText());
      if (!nextText) continue;
      if (reportTitlePattern.test(nextText)) {
        centered += centerParagraph_(paragraphs[next]);
      }
      break;
    }
  }

  Logger.log('[STOCK_PDF] Centered header paragraphs: ' + centered);
  return centered;
}

function normalizeStockPdfPreparedByBlock_(body) {
  if (!body) return 0;

  var started = false;
  var normalized = 0;

  for (var i = 0; i < body.getNumChildren(); i++) {
    var child = body.getChild(i);
    if (child.getType() !== DocumentApp.ElementType.PARAGRAPH) continue;

    var paragraph = child.asParagraph();
    var text = String(paragraph.getText() || '').replace(/\s+/g, ' ').trim();

    if (text.indexOf('จัดทำโดย :') === 0) started = true;
    if (!started) continue;

    if (
      text === '' ||
      text.indexOf('จัดทำโดย :') === 0 ||
      text === 'มหาวิทยาลัยสวนดุสิต ศูนย์การศึกษาลำปาง' ||
      text.indexOf('พัฒนาโดย :') === 0
    ) {
      try {
        paragraph.setAlignment(DocumentApp.HorizontalAlignment.LEFT)
          .setIndentStart(0)
          .setIndentFirstLine(0)
          .setFontFamily('Sarabun')
          .setFontSize(10.5)
          .setSpacingBefore(0)
          .setSpacingAfter(text === '' ? 7 : 4)
          .setLineSpacing(1.15);
        normalized++;
      } catch (e) {
        Logger.log('[STOCK_PDF] Prepared-by block normalize failed: ' + e.message);
      }
    }

    if (text.indexOf('พัฒนาโดย :') === 0) break;
  }

  return normalized;
}

function normalizeStockPdfStatusBlock_(body) {
  if (!body) return 0;

  var started = false;
  var normalized = 0;

  for (var i = 0; i < body.getNumChildren(); i++) {
    var child = body.getChild(i);
    if (child.getType() !== DocumentApp.ElementType.PARAGRAPH) continue;

    var paragraph = child.asParagraph();
    var text = String(paragraph.getText() || '').replace(/\s+/g, ' ').trim();

    if (text === 'สรุปสถานะอะไหล่') started = true;
    if (!started) continue;
    if (text === 'หมายเหตุ') break;

    if (text === '' || text.indexOf('━') >= 0) continue;

    try {
      paragraph.setFontFamily('Sarabun')
        .setFontSize(text === 'สรุปสถานะอะไหล่' ? 14 : 11)
        .setBold(text === 'สรุปสถานะอะไหล่')
        .setSpacingBefore(0)
        .setSpacingAfter(text === 'สรุปสถานะอะไหล่' ? 8 : 5)
        .setLineSpacing(1.15);
      normalized++;
    } catch (e) {
      Logger.log('[STOCK_PDF] Status block normalize failed: ' + e.message);
    }
  }

  return normalized;
}

function normalizeStockPdfNoteBlock_(body) {
  if (!body) return 0;

  var started = false;
  var normalized = 0;

  for (var i = 0; i < body.getNumChildren(); i++) {
    var child = body.getChild(i);
    if (child.getType() !== DocumentApp.ElementType.PARAGRAPH) continue;

    var paragraph = child.asParagraph();
    var text = String(paragraph.getText() || '').replace(/\s+/g, ' ').trim();

    if (text === 'หมายเหตุ') started = true;
    if (!started) continue;
    if (text.indexOf('จัดทำโดย :') === 0) break;

    if (text === '' || text.indexOf('━') >= 0) continue;

    try {
      paragraph.setFontFamily('Sarabun')
        .setFontSize(text === 'หมายเหตุ' ? 14 : 10.5)
        .setBold(text === 'หมายเหตุ')
        .setSpacingBefore(0)
        .setSpacingAfter(text === 'หมายเหตุ' ? 8 : 5)
        .setLineSpacing(1.15);
      normalized++;
    } catch (e) {
      Logger.log('[STOCK_PDF] Note block normalize failed: ' + e.message);
    }
  }

  return normalized;
}

function removeStockPdfFooterTitle_(doc) {

}

// [ANCHOR: SERVER-STOCK-TRIM-TRAILING-BLANK-PDF-PAGE]
function trimTrailingBlankStockPdfPage_(body) {
  if (!body) return 0;

  function isBlankParagraph_(paragraph) {
    if (!paragraph) return true;
    var text = String(paragraph.getText() || '').replace(/\u00A0/g, ' ').trim();
    if (text !== '') return false;

    for (var i = 0; i < paragraph.getNumChildren(); i++) {
      if (paragraph.getChild(i).getType() === DocumentApp.ElementType.INLINE_IMAGE) return false;
    }
    return true;
  }

  function isBlankTable_(table) {
    if (!table) return true;
    try {
      for (var r = 0; r < table.getNumRows(); r++) {
        var row = table.getRow(r);
        for (var c = 0; c < row.getNumCells(); c++) {
          var cell = row.getCell(c);
          for (var p = 0; p < cell.getNumChildren(); p++) {
            var child = cell.getChild(p);
            if (child.getType() === DocumentApp.ElementType.PARAGRAPH) {
              if (!isBlankParagraph_(child.asParagraph())) return false;
            } else if (child.getType() === DocumentApp.ElementType.INLINE_IMAGE) {
              return false;
            } else {
              return false;
            }
          }
        }
      }
      return true;
    } catch (e) {
      return false;
    }
  }

  var removed = 0;
  var safety = 0;

  // ลบตัวแบ่งหน้าทั้งเอกสาร เพราะ page break แฝงมักเป็นสาเหตุหน้าว่างท้าย PDF
  for (var i = body.getNumChildren() - 1; i >= 0; i--) {
    var element = body.getChild(i);
    var elementType = element.getType();
    if (elementType === DocumentApp.ElementType.PAGE_BREAK || elementType === DocumentApp.ElementType.SECTION_BREAK) {
      try {
        body.removeChild(element);
        removed++;
      } catch (removeBreakErr) {}
    }
  }

  // ลบย่อหน้าว่าง / ตารางว่างที่ท้ายเอกสารแบบไล่ย้อนจนเจอเนื้อหาจริง
  while (body.getNumChildren() > 0 && safety < 500) {
    safety++;

    var lastIndex = body.getNumChildren() - 1;
    var child = body.getChild(lastIndex);
    var type = child.getType();

    if (type === DocumentApp.ElementType.PARAGRAPH) {
      var paragraph = child.asParagraph();
      if (isBlankParagraph_(paragraph)) {
        try { paragraph.setPageBreakBefore(false); } catch (e1) {}
        try { paragraph.setSpacingBefore(0); } catch (e2) {}
        try { paragraph.setSpacingAfter(0); } catch (e3) {}
        try { paragraph.setLineSpacing(1); } catch (e4) {}
        try { paragraph.setFontSize(1); } catch (e5) {}
        try {
          body.removeChild(child);
          removed++;
          continue;
        } catch (removeErr) {
          break;
        }
      }
      break;
    }

    if (type === DocumentApp.ElementType.TABLE) {
      if (isBlankTable_(child.asTable())) {
        try {
          body.removeChild(child);
          removed++;
          continue;
        } catch (tableErr) {
          break;
        }
      }
      break;
    }

    if (type === DocumentApp.ElementType.PAGE_BREAK || type === DocumentApp.ElementType.SECTION_BREAK) {
      try {
        body.removeChild(child);
        removed++;
        continue;
      } catch (breakErr2) {
        break;
      }
    }

    break;
  }

  return removed;
}

// [ANCHOR: SERVER-STOCK-CREATE-NEW-PART]
function createNewSparePart(masterSheet, payload) {
  var newPart = payload.newPart || {};

  var partId = String(newPart.partId || '').trim();
  var partName = String(newPart.partName || '').trim();
  var category = String(newPart.category || '').trim();
  var partType = String(newPart.partType || 'วัสดุสิ้นเปลือง').trim() || 'วัสดุสิ้นเปลือง';
  var unit = String(newPart.unit || '').trim();
  var cost = Number(newPart.cost || 0);
  var reorderPoint = Number(newPart.reorderPoint || 0);
  var note = String(payload.note || '').trim();

  if (!partName || !category || !unit) {
    throw new Error('กรุณากรอกข้อมูลอะไหล่ใหม่ให้ครบ: ชื่ออะไหล่ หมวดหมู่ และหน่วยนับ');
  }

  if (cost < 0) {
    throw new Error('ราคาต่อหน่วยต้องไม่ติดลบ');
  }

  if (reorderPoint < 0) {
    throw new Error('จุดเตือนใกล้หมดต้องไม่ติดลบ');
  }

  if (partType !== 'วัสดุสิ้นเปลือง' && partType !== 'เครื่องมือ / อุปกรณ์') {
    throw new Error('ประเภทอะไหล่ไม่ถูกต้อง');
  }

  var masterData = masterSheet.getDataRange().getValues();
  var normalizedNewName = normalizePartName(partName);

  for (var i = 1; i < masterData.length; i++) {
    var existingPartId = String(masterData[i][0] || '').trim();
    var existingPartName = String(masterData[i][1] || '').trim();
    var normalizedExistingName = normalizePartName(existingPartName);

    if (partId && existingPartId === partId) {
      throw new Error('รหัสอะไหล่นี้มีอยู่แล้วในระบบ');
    }

    if (normalizedExistingName && normalizedExistingName === normalizedNewName) {
      throw new Error('ชื่ออะไหล่นี้มีอยู่แล้วในระบบ: ' + existingPartName);
    }
  }

  if (!partId) {
    partId = generatePartId(masterSheet);
  }

  var now = new Date();
  var d = String(now.getDate()).padStart(2, '0');
  var m = String(now.getMonth() + 1).padStart(2, '0');
  var y = now.getFullYear() + 543;
  var h = String(now.getHours()).padStart(2, '0');
  var min = String(now.getMinutes()).padStart(2, '0');
  var createdDate = d + '/' + m + '/' + y + ' ' + h + ':' + min + ' น.';
  var imageCol = ensureSparePartImageColumn_(masterSheet);
  var typeCol = ensureSparePartTypeColumn_(masterSheet);
  var imageUrl = uploadSparePartImage_(newPart.imageDataUrl || '', partId, newPart.imageFileName || '');

  var newRow = [
    partId,
    partName,
    category,
    unit,
    cost,
    0,
    reorderPoint,
    'ปกติ',
    createdDate,
    note
  ];
  while (newRow.length < Math.max(imageCol, typeCol)) newRow.push('');
  newRow[imageCol - 1] = imageUrl;
  newRow[typeCol - 1] = partType;

  masterSheet.appendRow(newRow);

  SpreadsheetApp.flush();

  return {
    partId: partId,
    partName: partName
  };
}

// [ANCHOR: SERVER-STOCK-PROCESS-TRANSACTION]
function processStockTransaction(payload) {
  var lock = LockService.getScriptLock();
  if (!lock.tryLock(15000)) {
    return { success: false, error: 'ระบบกำลังทำงาน กรุณารอสักครู่แล้วลองใหม่' };
  }

  try {
    Logger.log('STEP1 payload=' + JSON.stringify(payload));

    var ss = SpreadsheetApp.openById(CONFIG.SPREADSHEET_ID);
    var masterSheet = ss.getSheetByName('รายการอะไหล่ทั้งหมด');
    var ledgerSheet = ss.getSheetByName('ประวัติสต็อกอะไหล่');

    if (!masterSheet || !ledgerSheet) {
      throw new Error('โครงสร้างชีตสต็อกไม่สมบูรณ์');
    }

    var partId = String(payload.partId || '').trim();
    var type = String(payload.type || '').trim();
    var qty = Number(payload.qty || 0);
    var refJob = String(payload.refJob || '').trim();
    var user = String(payload.user || '').trim();
    var note = String(payload.note || '').trim();
    var isNewPart = (partId === '__new__');
    requireStockRole_(payload.currentUser, ['admin'], 'ทำรายการสต็อก');

    Logger.log('STEP2 isNewPart=' + isNewPart + ', type=' + type + ', qty=' + qty);

    if (qty <= 0 || !type) {
      throw new Error('ข้อมูลไม่ครบถ้วน หรือจำนวนไม่ถูกต้อง');
    }

    if (isNewPart && type !== 'รับเข้า') {
      throw new Error('การเพิ่มอะไหล่ใหม่ทำได้เฉพาะรายการรับเข้าเท่านั้น');
    }

    if (isNewPart) {
      var createdPart = createNewSparePart(masterSheet, payload);
      partId = createdPart.partId;
      Logger.log('STEP3 created new partId=' + partId);
    }

    var masterData = masterSheet.getDataRange().getValues();
    var rowIndex = -1;
    var currentBalance = 0;
    var partName = '';
    var cost = 0;
    var reorderPoint = 0;

    for (var i = 1; i < masterData.length; i++) {
      if (String(masterData[i][0] || '').trim() === partId) {
        rowIndex = i + 1;
        partName = String(masterData[i][1] || '').trim();
        cost = Number(masterData[i][4] || 0);
        currentBalance = Number(masterData[i][5] || 0);
        reorderPoint = Number(masterData[i][6] || 0);
        break;
      }
    }

    Logger.log('STEP4 rowIndex=' + rowIndex + ', currentBalance=' + currentBalance);

    if (rowIndex === -1) {
      throw new Error('ไม่พบรหัสอะไหล่ในระบบ');
    }

    var isDeduction = (type === 'เบิกออก' || type === 'ใช้ในงานซ่อม' || type === 'ปรับลด');
    var isAddition = (type === 'รับเข้า' || type === 'ปรับเพิ่ม' || type === 'คืนเข้า');

    var newBalance = currentBalance;

    if (isDeduction) {
      if (currentBalance < qty) {
        throw new Error('สต็อกไม่เพียงพอ (คงเหลือ: ' + currentBalance + ')');
      }
      newBalance = currentBalance - qty;
    } else if (isAddition) {
      newBalance = currentBalance + qty;
    } else {
      throw new Error('ประเภทรายการไม่ถูกต้อง');
    }

    Logger.log('STEP5 newBalance=' + newBalance);

    var totalPrice = mathMultiply(qty, cost);
    var now = new Date();
    var d = String(now.getDate()).padStart(2, '0');
    var m = String(now.getMonth() + 1).padStart(2, '0');
    var y = now.getFullYear() + 543;
    var h = String(now.getHours()).padStart(2, '0');
    var min = String(now.getMinutes()).padStart(2, '0');
    var dateStr = d + '/' + m + '/' + y + ' ' + h + ':' + min + ' น.';
    var transId = 'TR-' + y + m + d + '-' + String(now.getTime()).slice(-4);

    masterSheet.getRange(rowIndex, 6).setValue(newBalance);

    var stockStatus = newBalance <= reorderPoint ? 'ใกล้หมด' : 'ปกติ';
    masterSheet.getRange(rowIndex, 8).setValue(stockStatus);

    ledgerSheet.appendRow([
      transId,
      dateStr,
      partId,
      partName,
      type,
      qty,
      totalPrice,
      refJob,
      user,
      note
    ]);

    SpreadsheetApp.flush();

    Logger.log('STEP6 PASS processStockTransaction');

    return {
      success: true,
      message: 'บันทึกรายการสำเร็จ',
      newBalance: newBalance,
      isLowStock: newBalance <= reorderPoint,
      partId: partId,
      partName: partName
    };

  } catch (error) {
    Logger.log('FAIL processStockTransaction: ' + error.message);
    Logger.log(error.stack);
    return { success: false, error: error.message };
  } finally {
    lock.releaseLock();
  }
}

// [ANCHOR: SERVER-STOCK-GENERATE-MONTHLY-PDF]
function generateMonthlyStockReport(month, yearBE) {
  try {
    var ss = SpreadsheetApp.openById(CONFIG.SPREADSHEET_ID);
    var masterSheet = ss.getSheetByName('รายการอะไหล่ทั้งหมด');
    var ledgerSheet = ss.getSheetByName('ประวัติสต็อกอะไหล่');

    if (!masterSheet || !ledgerSheet) {
      throw new Error('ไม่พบชีตสำหรับสร้างรายงานสต็อก');
    }

    var tz = Session.getScriptTimeZone();
    var now = new Date();

    var thaiMonths = (CONFIG && CONFIG.MONTHS_TH && CONFIG.MONTHS_TH.length === 12)
      ? CONFIG.MONTHS_TH
      : ['มกราคม','กุมภาพันธ์','มีนาคม','เมษายน','พฤษภาคม','มิถุนายน','กรกฎาคม','สิงหาคม','กันยายน','ตุลาคม','พฤศจิกายน','ธันวาคม'];

    var currentYearBE = now.getFullYear() + 543;
    var targetMonth = Number(month || (now.getMonth() + 1));
    var targetYearBE = Number(yearBE || currentYearBE);
    var targetMonthText = String(targetMonth).padStart(2, '0');
    var monthLabel = thaiMonths[targetMonth - 1] + ' ' + targetYearBE;

    var reportDate =
      Utilities.formatDate(now, tz, 'd') + ' ' +
      thaiMonths[now.getMonth()] + ' ' +
      (now.getFullYear() + 543) + ' ' +
      Utilities.formatDate(now, tz, 'HH:mm') + ' น.';

    var fileMonthLabel = String(targetMonth).padStart(2, '0') + '-' + targetYearBE;

    var imageCol = 11; // Column K = รูปภาพอะไหล่
    var masterData = masterSheet.getDataRange().getValues();
    var ledgerData = ledgerSheet.getDataRange().getValues();

    var totalParts = 0;
    var lowStockCount = 0;
    var normalStockCount = 0;
    var totalStockValue = 0;
    var totalInQty = 0;
    var totalOutQty = 0;

    var bodyRows = [];
    var monthlyMovementMap = {};
    var activePartsById = {};
    var activePartOrder = [];

    for (var i = 1; i < masterData.length; i++) {
      var partId = String(masterData[i][0] || '').trim();
      var partName = String(masterData[i][1] || '').trim();
      var category = String(masterData[i][2] || '').trim();
      var unit = String(masterData[i][3] || '').trim();
      var cost = Number(masterData[i][4] || 0);
      var balance = Number(masterData[i][5] || 0);
      var reorderPoint = Number(masterData[i][6] || 0);
      var sheetStatus = String(masterData[i][7] || '').trim();
      var status = sheetStatus || (balance <= reorderPoint ? 'ใกล้หมด' : 'ปกติ');
      var partType = String(masterData[i][11] || '-').trim() || '-';
      var imageUrl = String(masterData[i][imageCol - 1] || '').trim();

      if (!partId && !partName) continue;
      if (!partId) continue;

      activePartsById[partId] = {
        partId: partId,
        partName: partName,
        category: category,
        unit: unit,
        cost: cost,
        balance: balance,
        reorderPoint: reorderPoint,
        status: status,
        partType: partType,
        imageUrl: imageUrl
      };
      activePartOrder.push(partId);

      totalParts++;
      if (status === 'ใกล้หมด') lowStockCount++;
      if (status !== 'ใกล้หมด') normalStockCount++;
      totalStockValue += (balance * cost);
    }

    for (var lm = 1; lm < ledgerData.length; lm++) {
      var mPartId = String(ledgerData[lm][2] || '').trim();
      var mType = String(ledgerData[lm][4] || '').trim();
      var mQty = Number(ledgerData[lm][5] || 0);
      var mDateText = String(ledgerData[lm][1] || '').trim();

      if (!mPartId || !activePartsById[mPartId]) continue;

      var mMatch = mDateText.match(/(\d{1,2})[\/.-](\d{1,2})[\/.-](\d{4})/);
      if (!mMatch) continue;

      var mMonth = mMatch[2].padStart(2, '0');
      var mYear = Number(mMatch[3]);
      if (mYear < 2400) mYear += 543;
      if (mMonth !== targetMonthText || mYear !== targetYearBE) continue;

            var src = String(text == null ? '' : text).replace(/\s+/g, ' ').trim();

                  lines.push(token.slice(i, i + tokenLimit));

                lines.push(currentLine);
                currentLine = token;
              } else {
                currentLine = nextLine;

// ANCHOR: ฟังก์ชันโหลดข้อมูลอะไหล่สำหรับนำไปแสดง... (หาบรรทัดนี้หรือใกล้เคียงเพื่อวางฟังก์ชันใหม่)
function getAvailableStockReportMonths() {
  try {
    var sheet = SpreadsheetApp.openById(CONFIG.SPREADSHEET_ID).getSheetByName('ประวัติสต็อกอะไหล่');
    var values = sheet.getRange(2, 2, Math.max(1, sheet.getLastRow() - 1), 1).getDisplayValues();
    var monthSet = new Set();
    var now = new Date();
    var currentYearBE = now.getFullYear() + 543;
    var currentMonth = now.getMonth() + 1;
    monthSet.add(currentYearBE + '-' + String(currentMonth).padStart(2, '0'));
    
    for (var i = 0; i < values.length; i++) {
      var dateStr = String(values[i][0] || '').trim();
      if (!dateStr) continue;
      var match = dateStr.match(/(\d{1,2})[\/.-](\d{1,2})[\/.-](\d{4})/);
      if (match) {
        var y = parseInt(match[3], 10);
        if (y < 2400) y += 543; // บังคับเป็น พ.ศ.
        var m = parseInt(match[2], 10);
        monthSet.add(y + '-' + String(m).padStart(2, '0'));
      }
    }
    
    var sorted = Array.from(monthSet).sort().reverse().map(function(key) {
      var parts = key.split('-');
      var y = parseInt(parts[0], 10);
      var m = parseInt(parts[1], 10);
      return { yearBE: y, month: m, label: CONFIG.MONTHS_TH[m - 1] + ' ' + y };
    });
    return { success: true, months: sorted };
  } catch (e) { return { success: false, error: e.message }; }
}

function formatShortDate(dateObj) {
  if (!(dateObj instanceof Date) || isNaN(dateObj)) return '-';
  var d = dateObj.getDate();
  var m = dateObj.getMonth() + 1;
  var y = dateObj.getFullYear();
  if (y < 2400) y += 543;
  return d + '/' + m + '/' + String(y).slice(-2); // คืนค่าแบบ 25/3/69
}

function updateSparePartRow(partId, newName, newCat, newUnit, newReorderPoint, imageDataUrl, imageFileName, newPartType, newBalance, currentUser, newCost) {
  var lock = LockService.getScriptLock();
  if (!lock.tryLock(10000)) return { success: false, error: 'ระบบยุ่ง กรุณาลองใหม่' };
  try {
    requireStockRole_(currentUser, ['admin'], 'แก้ไขข้อมูลอะไหล่');
    var sheet = SpreadsheetApp.openById(CONFIG.SPREADSHEET_ID).getSheetByName('รายการอะไหล่ทั้งหมด');
    var imageCol = ensureSparePartImageColumn_(sheet);
    var typeCol = ensureSparePartTypeColumn_(sheet);
    var data = sheet.getDataRange().getValues();
    for (var i = 1; i < data.length; i++) {
      if (String(data[i][0]).trim() === partId) {
        var partType = String(newPartType || data[i][typeCol - 1] || 'วัสดุสิ้นเปลือง').trim() || 'วัสดุสิ้นเปลือง';
        if (partType !== 'วัสดุสิ้นเปลือง' && partType !== 'เครื่องมือ / อุปกรณ์') {
          throw new Error('ประเภทอะไหล่ไม่ถูกต้อง');
        }

        var bal = Number(data[i][5] || 0);
        if (newBalance !== undefined && newBalance !== null && String(newBalance).trim() !== '') {
          bal = Number(newBalance);
        }
        if (isNaN(bal) || bal < 0) {
          throw new Error('ยอดคงเหลือปัจจุบันต้องเป็นตัวเลข 0 ขึ้นไป');
        }

        var costVal = Number(data[i][4] || 0);
        if (newCost !== undefined && newCost !== null && String(newCost).trim() !== '') {
          costVal = Number(newCost);
        }
        if (isNaN(costVal) || costVal < 0) {
          throw new Error('ราคาต่อหน่วยต้องเป็นตัวเลข 0 ขึ้นไป');
        }

        sheet.getRange(i + 1, 2).setValue(newName);
        sheet.getRange(i + 1, 3).setValue(newCat);
        sheet.getRange(i + 1, 4).setValue(newUnit);
        sheet.getRange(i + 1, 5).setValue(costVal);
        sheet.getRange(i + 1, 6).setValue(bal);
        sheet.getRange(i + 1, 7).setValue(newReorderPoint);
        sheet.getRange(i + 1, 8).setValue(bal <= Number(newReorderPoint) ? 'ใกล้หมด' : 'ปกติ');
        sheet.getRange(i + 1, typeCol).setValue(partType);
        if (imageDataUrl) {
          sheet.getRange(i + 1, imageCol).setValue(uploadSparePartImage_(imageDataUrl, partId, imageFileName || ''));
        }
        return { success: true };
      }
    }
    return { success: false, error: 'ไม่พบรหัสอะไหล่' };
  } catch(e) { return { success: false, error: e.message }; } finally { lock.releaseLock(); }
}

    assertTrue(typeCountMap['ใช้ในงานซ่อม'] >= 1, 'ledger ใช้ในงานซ่อมไม่ครบ');
    assertTrue(typeCountMap['คืนเข้า'] >= 1, 'ledger คืนเข้าไม่ครบ');
    assertTrue(typeCountMap['เบิกออก'] >= 1, 'ledger เบิกออกไม่ครบ');

    if (keepTestData) {

    } else {
      log('STEP10', 'DELETE PART');

      finalBalanceBeforeDelete: finalBalance,

    };

  } catch (error) {

      ledgerTypeCount: typeCountMap,
      createdDateInMaster: createdDateValue,

    };
  }
}

// [ANCHOR: SERVER-STOCK-TICKET-REPAIR-COSTS]
function getTicketRepairCosts(sequenceNumber) {
  try {
    const normalizeTicketNo = function(value) {
      return String(value || '').replace(/\s+/g, '').trim().toUpperCase();
    };
    const seq = normalizeTicketNo(sequenceNumber);
    if (!seq) {
      return { success: false, error: 'ไม่ระบุเลขที่แจ้งซ่อม' };
    }

    const sheet = SpreadsheetApp.openById(CONFIG.SPREADSHEET_ID).getSheetByName('ประวัติสต็อกอะไหล่');
    if (!sheet) {
      return { success: true, sequenceNumber: seq, totalItems: 0, grandTotal: 0, data: [] };
    }

    const lastRow = sheet.getLastRow();
    if (lastRow <= 1) {
      return { success: true, sequenceNumber: seq, totalItems: 0, grandTotal: 0, data: [] };
    }

    const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0].map(function(h) {
      return String(h || '').replace(/\s+/g, ' ').trim();
    });
    const idx = function(name) { return headers.indexOf(name); };
    const col = {
      partCode: idx('รหัสอะไหล่'),
      partName: idx('ชื่ออะไหล่'),
      type: idx('ประเภทรายการ'),
      qty: idx('จำนวน'),
      totalValue: idx('มูลค่ารวม'),
      refJob: idx('อ้างอิงเลขที่แจ้งซ่อม')
    };
    const required = ['partCode', 'partName', 'type', 'qty', 'totalValue', 'refJob'];
    for (let r = 0; r < required.length; r++) {
      if (col[required[r]] < 0) {
        return { success: false, error: 'ไม่พบ Header ที่จำเป็นในชีตประวัติสต็อกอะไหล่' };
      }
    }

    const values = sheet.getRange(2, 1, lastRow - 1, sheet.getLastColumn()).getValues();
    
    const matchedItems = [];
    let grandTotal = 0;

    for (let i = 0; i < values.length; i++) {
      const type = String(values[i][col.type] || '').trim();
      const refJob = normalizeTicketNo(values[i][col.refJob]);

      if (type === 'ใช้ในงานซ่อม' && refJob === seq) {
        const partCode = String(values[i][col.partCode] || '').trim();
        const partName = String(values[i][col.partName] || '').trim();
        
        // ป้องกันค่าว่าง, string, comma, หรือ NaN
        const qtyRaw = String(values[i][col.qty] || '0').replace(/,/g, '').trim();
        const totalValueRaw = String(values[i][col.totalValue] || '0').replace(/,/g, '').trim();
        
        const qty = Number(qtyRaw) || 0;
        const totalValue = Number(totalValueRaw) || 0;
        
        // ป้องกันจำนวนเป็น 0 หรือ ติดลบ
        if (qty > 0) {
          const costPerUnit = totalValue / qty;
          grandTotal += totalValue;

          matchedItems.push({
            partCode: partCode,
            partName: partName,
            qty: qty,
            costPerUnit: costPerUnit,
            totalValue: totalValue
          });
        }
      }
    }

    return {
      success: true,
      sequenceNumber: seq,
      totalItems: matchedItems.length,
      grandTotal: grandTotal,
      data: matchedItems
    };

  } catch (error) {
    Logger.log('Error in getTicketRepairCosts: ' + error.stack);
    return { success: false, error: error.message };
  }
}
